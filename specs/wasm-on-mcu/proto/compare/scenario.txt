# Same script for the PC (crab-sim test) and the board (compare.ps1).
# reset   param|in <index> <i|f> <value>   tick <steps> <dt_ms>   out (first 4 outputs as 32-bit hex)
# The letter is the value type: i = int/bool (i32), f = float (f32).
reset
param 0 f 0.5
in 0 f 1.0
in 1 f 0.25
in 2 i 7
tick 5 10
out
in 0 f -2.5
in 1 f 3.0
in 2 i -2147483648
tick 5 10
out
in 0 f 0.1
in 1 f 0.2
in 2 i 2147483647
param 0 f 3.3
tick 3 20
out
tick 40 7
out
